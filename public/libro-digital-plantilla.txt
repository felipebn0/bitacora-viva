function cuadro(ahora) {

  const cx0 = innerWidth / 2, cyShelf = estante.getBoundingClientRect().top + estante.offsetHeight / 2;

  const ex = 1 - Math.exp(-7 * dt);
  const objY = enCubierta ? clamp((mx - cx0) / (0.55 * d), -1, 1) * 9 : (mx - cx0) / innerWidth * 5;
  const objX = enCubierta ? -clamp((my - cyShelf) / (H / 2), -1, 1) * 6 : 0;
  tiltY += (objY - tiltY) * ex; tiltX += (objX - tiltX) * ex;
  const el = 1 - Math.exp(-12 * dt);
  const lim = innerWidth / 2 + 1.5 * d;
  for (const t of tapes) {
    const r = relativo(t.v);
    const x = F(r);
    if (Math.abs(x) > lim) { if (!t.oculto) { t.el.style.visibility = 'hidden'; t.oculto = true; } continue; }
    if (t.oculto) { t.el.style.visibility = ''; t.oculto = false; }
    const o = smooth(Math.max(0, 1 - Math.abs(r)));
    const objLift = hover === t.v && Math.abs(r) > .5 && !abierto ? 1 : 0;
    t.lift += (objLift - t.lift) * (reducir ? 1 : el);
    t.el.style.transform = `translate3d(${x.toFixed(2)}px, ${(-o * 0.025 * H - t.lift * 0.045 * H).toFixed(2)}px, ${(o * 0.2 * H).toFixed(2)}px) rotateZ(${(-4.5 * o).toFixed(3)}deg) rotateX(${(tiltX * o).toFixed(3)}deg) rotateY(${(-90 * o + tiltY * o).toFixed(3)}deg)`;

    t.sp.style.setProperty('--shade', (o * .55).toFixed(3));
    t.cv.style.setProperty('--shade', ((1 - o) * .5).toFixed(3));
  }
  // Anillo de abrir y cursor
  const objAbrir = abierto ? 0 : smooth(Math.max(0, 1 - Math.abs(T - s) * 3));
  abrirEscala += (objAbrir - abrirEscala) * (reducir ? 1 : 1 - Math.exp(-14 * dt));

  if (fino) {

    cursor.classList.toggle('spine', !enCubierta && hover >= 0 && Math.abs(relativo(hover)) > .5 && !abierto);

}
